        return new Promise((resolve) => {
            if (container.querySelector(selector)) return resolve(container.querySelector(selector));
            const observer = new MutationObserver(() => {
                if (container.querySelector(selector)) {
                    observer.disconnect(); resolve(container.querySelector(selector));
                }
            });

            setTimeout(() => { observer.disconnect(); resolve(null); }, timeout);
        });

    function setReactValue(element, value) {
        if (!element) return;
        if (element.hasAttribute('readonly')) element.removeAttribute('readonly');
        if (element.hasAttribute('disabled')) element.removeAttribute('disabled');

        const descElement = Object.getOwnPropertyDescriptor(element, 'value');
        const valueSetter = descElement ? descElement.set : undefined;
        
        const prototype = Object.getPrototypeOf(element);
        const descPrototype = Object.getOwnPropertyDescriptor(prototype, 'value');
        const prototypeValueSetter = descPrototype ? descPrototype.set : undefined;

        if (prototypeValueSetter && valueSetter !== prototypeValueSetter) {
            prototypeValueSetter.call(element, value);
        } else if (valueSetter) {
            valueSetter.call(element, value);
        } else {
            element.value = value;
        }
        element.dispatchEvent(new Event('input', { bubbles: true }));
        element.dispatchEvent(new Event('change', { bubbles: true }));
    }

    function parseSpanishNumber(str) {
        if (str === null || str === undefined) return 0;
        let limpio = str.toString().trim().replace(/[^0-9.,-]/g, '');
        if (limpio === '') return 0;
        if (limpio.includes('.') && limpio.includes(',')) {
            if (limpio.indexOf('.') < limpio.indexOf(',')) limpio = limpio.replace(/\./g, '').replace(',', '.');
            else limpio = limpio.replace(/,/g, '');
        }
        else if (limpio.includes(',')) limpio = limpio.replace(',', '.');
        else if (limpio.includes('.')) {
            let partes = limpio.split('.');
            let decimales = partes[partes.length - 1];
            if (decimales.length === 3) limpio = limpio.replace(/\./g, '');
        }
        return parseFloat(limpio) || 0;
    }

        if (!elemento) return 0;

        if (ariaVal !== null && ariaVal !== "") return parseFloat(ariaVal) || 0;
        return parseSpanishNumber(elemento.value);
    }

    function procesarTextoPegado(texto) {
        let mapa = {}; let mapaMulti = {}; let noEncontrados = new Set(); let duplicados = new Set();
        texto.trim().split('\n').forEach(linea => {
            let l = linea.trim(); if (!l) return;
            let p = l.includes('\t') ? l.split('\t') : l.split(/\s{2,}/);
            if (p.length < 2 && l.lastIndexOf(' ') !== -1) {
                p = [l.substring(0, l.lastIndexOf(' ')), l.substring(l.lastIndexOf(' ') + 1)];
            }
            if (p.length >= 2) {
                let ref = p[0].trim();
                if (ref.toLowerCase() !== "ref." && ref.toLowerCase() !== "part number" && ref !== "") {
                    let clave = ref.toUpperCase(); let valor = parseSpanishNumber(p[p.length - 1]);
                    if (mapa[clave] !== undefined && mapa[clave] !== valor) duplicados.add(ref);
                    mapa[clave] = valor;
                    if (!mapaMulti[clave]) mapaMulti[clave] = [];
                    mapaMulti[clave].push(valor);
                    noEncontrados.add(clave);
                }
            }
        });
        return { mapa, mapaMulti, noEncontrados, duplicados };
    }

    function convertirAHTML(texto) {
        if (!texto) return "";
        return texto.split('\n').map(l => {
            l = l.trim(); if (!l) return "";
            l = l.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
            if (/^[\-\*•]\s+/.test(l)) l = l.replace(/^[\-\*•]\s+/, '&bull; ');
            // Uso estandar sin $ por si se pierden en el copiado
            l = l.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>').replace(/_(.*?)_/g, '<em>$1</em>').replace(/\*(.*?)\*/g, '<em>$1</em>');
            return l;
        }).filter(l => l.length > 0).join(' <br> ');
    }

    function parseMultipleGenerics(text) {
        let lines = text.split('\n'); let items = []; let currentItem = null;
        for (let line of lines) {
            let trimmed = line.trim();
            if (trimmed.includes('@@')) {
                if (currentItem) items.push(currentItem);
                let parts = trimmed.split('@@');
                currentItem = { pn: parts[0].trim(), descLines: [parts.slice(1).join('@@')] };
            } else if (currentItem && trimmed !== '') {
                currentItem.descLines.push(trimmed);
            }
        }
        if (currentItem) items.push(currentItem);
        return items;
    }

    optValMult.onclick = () => {
        if (baseDatosMult.size === 0) {
            mostrarToast("[!] La BD de Múltiplos aún no carga. Reintentando...", false);
            cargarBaseDatosMult();
            return;
        }

        let filas = document.querySelectorAll('tbody tr');
        let procesados = 0; let errores = 0;

        for (let idx = 0; idx < filas.length; idx++) {
            let fila = filas[idx];
            let idFinal = getIdentificadorFila(fila);
            if (!idFinal) continue;

            let mult = baseDatosMult.get(idFinal) || 1;
            let inputStock = fila.querySelector('input[data-test-id*="stock_pe"]');
            let inputQty = getInputCantidad(fila);
            if (!inputQty) continue;

            let stock = inputStock ? obtenerValorNumerico(inputStock) : 0;
            let qty = obtenerValorNumerico(inputQty);
            let esValido = true; let mensajeError = "";

            if (qty > stock) {
                let faltante = qty - stock;
                if (faltante % mult !== 0) {
                    esValido = false;
                    if (stock === 0) mensajeError = "Aumentar a múltiplo de " + mult;
                    else mensajeError = "Pedir " + stock + " (Stock) + mult. de " + mult;
                }
            }

            fila.querySelectorAll('.alerta-mult-badge').forEach(b => b.remove());
            if (fila.style.backgroundColor === 'rgb(254, 226, 226)' || fila.style.backgroundColor === COLOR_ERROR) {
                fila.style.backgroundColor = "";
                fila.querySelectorAll('td').forEach(td => { td.style.backgroundColor = ""; });
            }

            let celdaNombre = fila.querySelector('[data-test-id="line-item-name-cell"], [data-selenium-test="line-item-name-cell"]');
            if (esValido) {
                if (mult > 1 && celdaNombre) {
                    let badge = document.createElement('span');
                    badge.className = 'alerta-mult-badge'; badge.style.backgroundColor = '#00bda5'; badge.innerText = "Mult: " + mult + " OK";
                    celdaNombre.appendChild(badge);
                }
                procesados++;
            } else {
                fila.style.backgroundColor = COLOR_ERROR;
                fila.querySelectorAll('td').forEach(td => { td.style.backgroundColor = COLOR_ERROR; });
                if (celdaNombre) {
                    let badge = document.createElement('span');
                    badge.className = 'alerta-mult-badge'; badge.style.backgroundColor = '#ff3333'; badge.innerText = "[!] " + mensajeError;
                    celdaNombre.appendChild(badge);
                }
                errores++;
            }
        }

        if (errores === 0) mostrarToast("[OK] " + procesados + " productos validados. ¡Todo perfecto!");
        else mostrarToast("[!] Revisión completada: " + errores + " filas requieren atención.", false);
    };

                    </div>`;
            }).join('');

            const modalDiv = document.createElement('div');
            modalDiv.className = 'modal-precision';
            modalDiv.innerHTML = '<div class="modal-content-precision" style="border-top: 5px solid #ff7a59; max-width: 600px;">' +
                '<h3 style="margin-top:0; color:#ff7a59;">[!] PN repetido en la tabla</h3>' +
                '<p style="font-size:13px; color:#33475b; margin-bottom:10px;"><b>' + pnSeguro + '</b> aparece <b>' + filasPN.length + ' veces</b>. Indica ' + (esPrecio ? 'el precio' : 'la cantidad') + ' de cada posición:</p>' +
                '<div style="max-height: 300px; overflow-y: auto; border: 1px solid #cbd6e2; border-radius: 4px; padding: 8px 12px; margin-bottom: 12px; background: #f8fafc;">' + filasHTML + '</div>' +
                '<div id="divValidacionSplit" style="padding:8px 12px; border-radius:4px; font-size:12px; font-weight:bold; margin-bottom:12px;"></div>' +
                '<div style="display:flex; justify-content:flex-end; gap:10px;">' +
                    '<button id="btnCancelarSplit" style="padding:8px 16px; background:#eaf0f6; border:none; border-radius:4px; cursor:pointer;">Cancelar</button>' +
                    '<button id="btnAceptarSplit" style="padding:8px 16px; background:#ff7a59; color:white; border:none; border-radius:4px; cursor:pointer; font-weight:bold;">Aplicar</button>' +
                '</div>' +
            '</div>';
            document.body.appendChild(modalDiv);

            const divValidacion = modalDiv.querySelector('#divValidacionSplit');
            const btnAceptar = modalDiv.querySelector('#btnAceptarSplit');
            const inputs = Array.from(modalDiv.querySelectorAll('.div-pos-input'));

            function validar() {
                let ok = false, msg = '', estado = 'error';
                if (!inputs.every(i => i.value.trim() !== '')) {
                    msg = 'Completa todas las posiciones.';
                } else if (totalExacto !== null && totalExacto !== undefined) {
                    const suma = inputs.reduce((a, i) => a + parseSpanishNumber(i.value), 0);
                    const dif = totalExacto - suma;
                    if (Math.abs(dif) > 0.0001) {
                        msg = dif > 0 ? `Faltan ${+dif.toFixed(2)} (total ${totalExacto})` : `Sobran ${+(-dif).toFixed(2)} (total ${totalExacto})`;
                        estado = 'alerta';
                    } else { msg = `Suma correcta: ${totalExacto}`; ok = true; estado = 'exito'; }
                } else { msg = `${filasPN.length} posiciones listas.`; ok = true; estado = 'exito'; }

                const paleta = { error: [COLOR_ERROR, '#ff3333'], alerta: [COLOR_ALERTA, '#e59a10'], exito: [COLOR_EXITO, '#00bda5'] }[estado];
                divValidacion.style.background = paleta[0]; divValidacion.style.color = paleta[1]; divValidacion.textContent = msg;
                btnAceptar.disabled = !ok; btnAceptar.style.opacity = ok ? '1' : '0.5'; btnAceptar.style.cursor = ok ? 'pointer' : 'not-allowed';
                return ok;
            }

            const cerrar = (valor) => { modalDiv.remove(); resolve(valor); };
            inputs.forEach(inp => inp.addEventListener('input', validar));
            modalDiv.addEventListener('keydown', (e) => {
                if (e.key === 'Escape') { e.stopPropagation(); cerrar(null); }
                else if (e.key === 'Enter' && validar()) { e.preventDefault(); btnAceptar.click(); }
            });
            btnAceptar.onclick = () => { if (validar()) cerrar(inputs.map(i => parseSpanishNumber(i.value))); };
            modalDiv.querySelector('#btnCancelarSplit').onclick = () => cerrar(null);

            validar();
            (inputs.find(i => i.value.trim() === '') || inputs[0]).focus();
        });
    }

                if (bloqueadasGlobal.length) avisos.push(`Valor inválido: ${bloqueadasGlobal.map(b => b.pn).join(', ')}`);
                if (noAplicados.length) avisos.push(`No aplicados (Fila movida/borrada): ${noAplicados.join(', ')}`);

                if (avisos.length) mostrarToast("[!] " + avisos.join('\n'), false);
                else flash(btnSuite);
                return;
            }

            // --- FASE 1: PARSEO Y REPORTE ---
            const { mapaMulti } = procesarTextoPegado(document.getElementById("txtDatosSuite").value);
            const claves = Object.keys(mapaMulti);
            if (claves.length === 0) { mostrarToast("[-] Datos no válidos.", false); return; }

            procesando = true;
            btnEjecutar.innerHTML = "[...] Procesando..."; btnEjecutar.disabled = true;

            filasAActualizarGlobal = [];
            noEncontradosGlobal = [];
            bloqueadasGlobal = [];

            const filasPorPN = {};
            for (const fila of document.querySelectorAll('tbody tr')) {
                const idFinal = getIdentificadorFila(fila);
                if (!idFinal) continue;
                const iQty = getInputCantidad(fila);
                const iPrice = fila.querySelector(SEL_PRECIO);
                if (!iQty || !iPrice) continue;
                const clave = idFinal.trim().toUpperCase();
                (filasPorPN[clave] = filasPorPN[clave] || []).push({
                    fila, idFinal,
                    iDiscount: fila.querySelector(SEL_DESC),
                    iTiempo: fila.querySelector(SEL_TIEMPO),
                    cant: obtenerValorNumerico(iQty),
                    pOrig: obtenerValorNumerico(iPrice)
                });
            }

            const noEncontradosIniciales = claves.filter(c => !filasPorPN[c]);
            noEncontradosGlobal = noEncontradosIniciales;
            const mapaFinal = {};
            const erroresDivision = [];

            for (const clave of claves) {
                const filasPN = filasPorPN[clave];
                if (!filasPN) continue;
                const vals = mapaMulti[clave];
                const N = filasPN.length, M = vals.length;
                const pn = filasPN[0].idFinal;

                if (M === N) { mapaFinal[clave] = vals.slice(); continue; }

                if (tipoLogica === 'descuento' || tipoLogica === 'tiempo') {
                    if (vals.every(v => v === vals[0])) mapaFinal[clave] = Array(N).fill(vals[0]);
                    else erroresDivision.push({ pn, razon: `Pegaste ${M} valores distintos y hay ${N} filas.` });
                    continue;
                }

                if (M > N) { erroresDivision.push({ pn, razon: `Pegaste ${M} valores y hay ${N} filas.` }); continue; }
                if (tipoLogica === 'precio' && M === 1) { mapaFinal[clave] = Array(N).fill(vals[0]); continue; }

                const resModal = await pedirDivisionPorPosicion(
                    pn,
                    tipoLogica === 'cant' ? 'cantidad' : 'precio',
                    filasPN.map(i => ({ cantActual: i.cant, precioActual: i.pOrig })),
                    vals,
                    (tipoLogica === 'cant' && M === 1) ? vals[0] : null
                );
                if (resModal === null) { btnEjecutar.innerHTML = titulos[tipoLogica].btn; btnEjecutar.disabled = false; procesando = false; return; }
                mapaFinal[clave] = resModal;
            }

            if (erroresDivision.length > 0) {
                const modalErr = document.createElement('div');
                modalErr.className = 'modal-precision';
                const listaHTML = erroresDivision.map(e =>
                    '<div style="padding:8px; border-bottom:1px solid #eaf0f6;"><b>' + escapeHTML(e.pn) + '</b><br><span style="color:#ff3333; font-size:12px;">' + escapeHTML(e.razon) + '</span></div>'
                ).join('');
                modalErr.innerHTML = '<div class="modal-content-precision" style="border-top: 5px solid #ff3333; max-width: 500px;">' +
                    '<h3 style="margin-top:0; color:#ff3333;">[X] Datos inconsistentes</h3>' +
                    '<div style="max-height: 250px; overflow-y: auto; border: 1px solid #cbd6e2; border-radius: 4px; padding: 8px; margin-bottom: 15px;">' + listaHTML + '</div>' +
                    '<div style="display:flex; justify-content:flex-end;">' +
                        '<button id="btnCerrarErrDiv" style="padding:8px 16px; background:#eaf0f6; border:none; border-radius:4px; cursor:pointer;">Entendido</button>' +
                    '</div>' +
                '</div>';
                document.body.appendChild(modalErr);
                document.getElementById('btnCerrarErrDiv').onclick = () => { modalErr.remove(); btnEjecutar.innerHTML = titulos[tipoLogica].btn; btnEjecutar.disabled = false; procesando = false; };
                return;
            }

            const res = [];
            const campo = tipoLogica === 'descuento' ? 'desc' : tipoLogica;

            if (tipoLogica === 'descuento') res.push("PN\tCant\tPrecio_Orig\t%Desc\tPrecio_Desc\tTotal\tAlerta");
            else if (tipoLogica === 'cant') res.push("PN\tCant_Anterior\tCant_Nueva\tAlerta");
            else if (tipoLogica === 'precio') res.push("PN\tPrecio_Anterior\tPrecio_Nuevo\tAlerta");
            else if (tipoLogica === 'tiempo') res.push("PN\tTiempo_Anterior\tTiempo_Nuevo\tAlerta");

            const formatNum = (num) => (num !== "" && num !== undefined) ? Number(num).toFixed(2).replace('.', ',') : "";

            for (const clave in filasPorPN) {
                const filasPN = filasPorPN[clave];
                const valores = mapaFinal[clave];
                if (valores === undefined && tipoLogica !== 'descuento' && tipoLogica !== 'tiempo') continue;

                for (let idx = 0; idx < filasPN.length; idx++) {
                    const item = filasPN[idx];
                    const { fila, iDiscount, iTiempo, cant, pOrig, idFinal } = item;
                    const valorMatch = valores !== undefined ? valores[idx] : undefined;
                    const hayValor = valorMatch !== undefined;
                    let alerta = "";
                    let bloquear = false;

                    if (hayValor) {
                        if (tipoLogica === 'descuento') {
                            if (valorMatch < 0) { bloquear = true; alerta += "Desc. negativo "; }
                            if (valorMatch > 100) { bloquear = true; alerta += "Desc. > 100% "; }
                            if (valorMatch > VAL_DESC_MAX_SOSPECHOSO && valorMatch <= 100) alerta += "Desc. inusual ";
                        } else if (tipoLogica === 'cant') {
                            if (valorMatch < 0) { bloquear = true; alerta += "Cant. negativa "; }
                            if (valorMatch % 1 !== 0) alerta += "Cant. decimal ";
                            if (valorMatch > VAL_CANT_MAX) alerta += "Cant. alta ";
                        } else if (tipoLogica === 'precio') {
                            if (valorMatch < 0) { bloquear = true; alerta += "Precio negativo "; }
                            if (valorMatch > VAL_PRECIO_MAX) alerta += "Precio alto ";
                        } else if (tipoLogica === 'tiempo') {
                            if (valorMatch < 0) { bloquear = true; alerta += "Tiempo negativo "; }
                        }
                    }

                    if (bloquear) {
                        bloqueadasGlobal.push({ pn: idFinal, msg: alerta });
                        res.push(`${idFinal}\t---\t---\t[-] ${alerta}`);
                        pintarFila(fila, COLOR_ERROR);
                        continue; 
                    }

                    const cambio = { clave, pos: idx, campo, fila, pn: idFinal, valorFloat: hayValor ? parseFloat(valorMatch) : null };

                    if (tipoLogica === 'descuento') {
                        let pNuevo = ""; let pTotal = cant * pOrig; let descAplicado = 0;
                        if (hayValor) {
                            descAplicado = Math.min(Math.max(parseFloat(valorMatch) || 0, 0), 100);
                            pNuevo = pOrig * (1 - (descAplicado / 100)); pTotal = cant * pNuevo;
                            cambio.valorFloat = descAplicado;
                            if (iDiscount) filasAActualizarGlobal.push(cambio);
                            else alerta += "Sin campo desc ";
                        }
                        res.push(`${idFinal}\t${cant}\t${formatNum(pOrig)}\t${descAplicado.toString().replace('.', ',')}%\t${formatNum(pNuevo)}\t${formatNum(pTotal)}\t${alerta}`);
                    } else if (tipoLogica === 'cant') {
                        filasAActualizarGlobal.push(cambio);
                        res.push(`${idFinal}\t${cant}\t${valorMatch}\t${alerta}`);
                    } else if (tipoLogica === 'precio') {
                        filasAActualizarGlobal.push(cambio);
                        res.push(`${idFinal}\t${formatNum(pOrig)}\t${formatNum(valorMatch)}\t${alerta}`);
                    } else if (tipoLogica === 'tiempo') {
                        if (iTiempo) filasAActualizarGlobal.push(cambio);
                        else alerta += "Sin campo tiempo ";
                        let tOrig = iTiempo ? obtenerValorNumerico(iTiempo) : 0;
                        res.push(`${idFinal}\t${tOrig}\t${valorMatch}\t${alerta}`);
                    }
                } 
            }

            const encabezadoNoEnc = noEncontradosGlobal.length > 0 ? `[!] NO ENCONTRADOS: ${noEncontradosGlobal.join(', ')}\n\n` : "";

            btnEjecutar.style.width = "48%";
            btnEjecutar.innerHTML = "Agregar Cambios"; btnEjecutar.disabled = false;
            const btnCopiar = document.getElementById("btnCopiarReporteMasivo");
            btnCopiar.style.display = "block";

            document.getElementById("txtDatosSuite").style.display = "none";
            const txtReporte = document.getElementById("txtReporteOculto");
            txtReporte.style.cssText = "display:block; width:100%; height:200px; padding:10px; border:1px solid " + (noEncontradosGlobal.length ? "#ff3333" : "#cbd6e2") + "; border-radius:4px; font-size:11px; box-sizing:border-box; font-family:monospace; white-space: pre; background-color:#f8fafc;";
            txtReporte.value = encabezadoNoEnc + res.join('\n');

            btnCopiar.onclick = () => {
                txtReporte.select();
                try { GM_setClipboard(txtReporte.value); } catch (e) {}
                btnCopiar.innerHTML = "[OK] Copiado"; btnCopiar.style.backgroundColor = "#00bda5"; btnCopiar.style.color = "white";
                setTimeout(() => { btnCopiar.innerHTML = "Copiar Reporte"; btnCopiar.style.backgroundColor = "#f5f8fa"; btnCopiar.style.color = "#33475b"; }, 2000);
            };

            procesando = false;
        };
    }

                    fila.dispatchEvent(new MouseEvent('mouseover', { bubbles: true }));
                    await esperar(150);

                    let btnAcciones = fila.querySelector('button[data-test-id="library-item-dropdown-actions-btn"]') || fila.querySelector('button[data-selenium-test="library-item-dropdown-actions-btn"]');

                        await esperar(300);
                    }
                }
            }

            let noEncontrados = [...pnsSet].filter(pn => !pnsEncontrados.has(pn));

            if (noEncontrados.length > 0) mostrarToast(`[!] No encontrados:\n${noEncontrados.join(', ')}`, false);
            else flash(btnSuite);
        };
